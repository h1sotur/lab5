# Некоректний шлях: 404
curl.exe -i http://127.0.0.1:3000/abc
curl.exe -i http://127.0.0.1:3000/2000

# Недозволені методи: 405 і заголовок Allow
curl.exe -i -X POST http://127.0.0.1:3000/200
curl.exe -i -X PATCH http://127.0.0.1:3000/200

# GET: картинки немає в кеші, X-Cache: MISS, файл з'являється в теці кешу
curl.exe -s -D - -o NUL http://127.0.0.1:3000/418

# GET: повторний запит, X-Cache: HIT
curl.exe -s -D - -o NUL http://127.0.0.1:3000/418

# GET: код, якого немає на http.cat, відповідь 404, у кеш нічого не потрапляє
curl.exe -i http://127.0.0.1:3000/999
